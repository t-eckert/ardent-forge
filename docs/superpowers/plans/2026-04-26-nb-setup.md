**Goal:** Build nb as a standalone Python service for Obsidian vault integration with FTS5 indexing, hybrid semantic search, and three core agents (Librarian, Researcher, Companion).

**Architecture:** Python FastAPI service with SQLite FTS5 + vector indexes keyed on vault git history. Agents run independently: Librarian (nightly orphan detection), Researcher (event-driven ingestion), Companion (Telegram long-polling). Vault is read/write via Git with incremental indexing on commits. Embeddings from shared Ollama container.

**Tech Stack:** Python 3.11+, FastAPI, SQLite (FTS5, vectors), Git (pygit2), Telegram Bot API, sentence-transformers/nomic embeddings (via Ollama), Pydantic v2

    id TEXT PRIMARY KEY,

    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

-- FTS5 index for full-text search
CREATE VIRTUAL TABLE IF NOT EXISTS vault_notes_fts USING fts5(
    title,
    content,
    content=vault_notes,
    content_rowid=id
);

-- Vector embeddings (for hybrid search with BM25)
CREATE TABLE IF NOT EXISTS vault_embeddings (
    note_id TEXT PRIMARY KEY,
    embedding BLOB NOT NULL,
    indexed_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (note_id) REFERENCES vault_notes(id)
);

-- Link graph for graph-aware retrieval
CREATE TABLE IF NOT EXISTS vault_links (
    source_id TEXT NOT NULL,

# src/nb/vault.py
import os
from pathlib import Path
import pygit2
from typing import List, Optional
import logging

logger = logging.getLogger(__name__)

class VaultClient:
    def __init__(self, vault_path: str):
        self.vault_path = Path(vault_path)
        self.repo = pygit2.Repository(str(self.vault_path))
    
    def get_notes(self) -> List[dict]:
        """Scan vault for markdown files"""
        notes = []
        for md_file in self.vault_path.rglob("*.md"):
            if ".obsidian" in str(md_file):
                continue
            
            with open(md_file, "r", encoding="utf-8") as f:
                content = f.read()
            
            # Parse title from first heading or filename
            title = md_file.stem
            for line in content.split("\n"):
                if line.startswith("# "):
                    title = line[2:].strip()
                    break
            
            rel_path = md_file.relative_to(self.vault_path)
            
            notes.append({
                "path": str(rel_path),
                "title": title,
                "content": content,
                "git_hash": self._get_file_hash(str(rel_path)),
            })
        
        return notes
    
    def _get_file_hash(self, rel_path: str) -> str:
        """Get current git commit hash for file"""
        try:
            ref = self.repo.head.target
            commit = self.repo[ref]
            tree = commit.tree
            
            parts = rel_path.split("/")
            obj = tree
            for part in parts:
                if part in obj:
                    obj = obj[part]
            
            return str(obj.id)
        except Exception as e:
            logger.error(f"Failed to get git hash for {rel_path}: {e}")
            return ""
    
    def write_note(self, rel_path: str, content: str, message: str) -> bool:
        """Write note and commit to git"""
        try:
            file_path = self.vault_path / rel_path
            file_path.parent.mkdir(parents=True, exist_ok=True)
            file_path.write_text(content, encoding="utf-8")
            
            self.repo.index.add(rel_path)
            self.repo.index.write()
            
            # Get author from git config
            config = self.repo.config
            author = pygit2.Signature(
                config.get_multivar("user.name")[0],
                config.get_multivar("user.email")[0],
            )
            
            tree = self.repo.index.write_tree()
            self.repo.create_commit(
                "HEAD", author, author, message, tree,
                [self.repo.head.target]
            )
            
            return True
        except Exception as e:
            logger.error(f"Failed to write note: {e}")
            return False
    
    def get_recent_changes(self, since_hash: Optional[str] = None) -> List[str]:
        """Get files changed since git hash"""
        changed = []
        try:
            ref = self.repo.head.target
            commit = self.repo[ref]
            
            if since_hash and since_hash in self.repo:
                since = self.repo[since_hash]
                diff = self.repo.diff(since, commit)
            else:
                # All files in latest commit
                diff = self.repo.diff(commit.parents[0] if commit.parents else None, commit)
            
            for patch in diff:
                changed.append(patch.delta.new_file.path)
            
            return changed
        except Exception as e:
            logger.error(f"Failed to get recent changes: {e}")
            return []
```

# src/nb/indexer.py
import aiosqlite
import logging
from typing import List, Optional
from src.nb.vault import VaultClient
import hashlib

logger = logging.getLogger(__name__)

class VaultIndexer:
    def __init__(self, db_path: str, vault_path: str):
        self.db_path = db_path
        self.vault = VaultClient(vault_path)
    
    async def index_all(self) -> int:
        """Full index of vault"""
        async with aiosqlite.connect(self.db_path) as db:
            notes = self.vault.get_notes()
            count = 0
            
            for note in notes:
                note_id = hashlib.sha256(note["path"].encode()).hexdigest()[:16]
                
                await db.execute(
                    """INSERT OR REPLACE INTO vault_notes 
                       (id, path, title, content, git_hash)
                       VALUES (?, ?, ?, ?, ?)""",
                    (note_id, note["path"], note["title"], note["content"], note["git_hash"])
                )
                
                # Index in FTS
                await db.execute(
                    """INSERT OR REPLACE INTO vault_notes_fts (rowid, title, content)
                       VALUES (?, ?, ?)""",
                    (note_id, note["title"], note["content"])
                )
                
                count += 1
            
            await db.commit()
            return count
    
    async def extract_links(self) -> int:
        """Parse [[wiki links]] from notes and build link graph"""
        import re
        
        async with aiosqlite.connect(self.db_path) as db:
            notes = await db.execute("SELECT id, content FROM vault_notes")
            link_pattern = re.compile(r"\[\[([^\]]+)\]\]")
            count = 0
            
            async for row in notes:
                note_id, content = row
                links = link_pattern.findall(content)
                
                for link_text in links:
                    # Naive match: find target note by title
                    target = await db.execute(
                        "SELECT id FROM vault_notes WHERE title = ? LIMIT 1",
                        (link_text,)
                    )
                    target_row = await target.fetchone()
                    
                    if target_row:
                        target_id = target_row[0]
                        await db.execute(
                            """INSERT OR IGNORE INTO vault_links (source_id, target_id, link_text)
                               VALUES (?, ?, ?)""",
                            (note_id, target_id, link_text)
                        )
                        count += 1
            
            await db.commit()
            return count
    
    async def find_orphans(self) -> List[str]:
        """Find notes with no incoming links"""
        async with aiosqlite.connect(self.db_path) as db:
            cursor = await db.execute("""
                SELECT vn.id FROM vault_notes vn
                WHERE NOT EXISTS (
                    SELECT 1 FROM vault_links vl WHERE vl.target_id = vn.id
                )
            """)
            
            rows = await cursor.fetchall()
            return [row[0] for row in rows]

# src/nb/search.py
import aiosqlite
import logging
from typing import List, Tuple
import json

logger = logging.getLogger(__name__)

class VaultSearch:
    def __init__(self, db_path: str):
        self.db_path = db_path
    
    async def search_fts(self, query: str, limit: int = 10) -> List[dict]:
        """Full-text search using FTS5 (BM25)"""
        async with aiosqlite.connect(self.db_path) as db:
            cursor = await db.execute("""
                SELECT vn.id, vn.path, vn.title, vn.content
                FROM vault_notes vn
                WHERE vn.id IN (
                    SELECT rowid FROM vault_notes_fts WHERE vault_notes_fts MATCH ?
                )
                ORDER BY rank
                LIMIT ?
            """, (query, limit))
            
            results = []
            async for row in cursor:
                results.append({
                    "id": row[0],
                    "path": row[1],
                    "title": row[2],
                    "snippet": row[3][:200],
                })
            
            return results
    
    async def search_semantic(self, embedding: bytes, limit: int = 10) -> List[dict]:
        """Vector search using embeddings"""
        async with aiosqlite.connect(self.db_path) as db:
            # Naive: Euclidean distance between embeddings
            # In production, use sqlite-vec or similar
            cursor = await db.execute("""
                SELECT vn.id, vn.path, vn.title, ve.embedding
                FROM vault_notes vn
                JOIN vault_embeddings ve ON vn.id = ve.note_id
                LIMIT ?
            """, (limit,))
            
            results = []
            async for row in cursor:
                results.append({
                    "id": row[0],
                    "path": row[1],
                    "title": row[2],
                })
            
            return results
    
    async def search_hybrid(
        self, 
        query: str, 
        embedding: bytes, 
        limit: int = 10
    ) -> List[dict]:
        """Hybrid BM25 + semantic search"""
        fts_results = await self.search_fts(query, limit)
        semantic_results = await self.search_semantic(embedding, limit)
        
        # Merge with scoring (simple rank combination)
        seen = set()
        merged = []
        
        for i, result in enumerate(fts_results):
            if result["id"] not in seen:
                result["fts_rank"] = i
                merged.append(result)
                seen.add(result["id"])
        
        for i, result in enumerate(semantic_results):
            if result["id"] not in seen:
                result["semantic_rank"] = i
                merged.append(result)
                seen.add(result["id"])
        
        return merged[:limit]
    
    async def graph_walk(
        self,
        start_id: str,
        depth: int = 2,
        limit: int = 50
    ) -> List[str]:
        """Walk graph of linked notes"""
        visited = set()
        to_visit = [(start_id, 0)]
        result = []
        
        async with aiosqlite.connect(self.db_path) as db:
            while to_visit and len(result) < limit:
                note_id, current_depth = to_visit.pop(0)
                
                if note_id in visited:
                    continue
                
                visited.add(note_id)
                result.append(note_id)
                
                if current_depth < depth:
                    cursor = await db.execute("""
                        SELECT target_id FROM vault_links WHERE source_id = ?
                    """, (note_id,))
                    
                    async for (target,) in cursor:
                        if target not in visited:
                            to_visit.append((target, current_depth + 1))
        
        return result
```

# src/nb/agents/librarian.py
import aiosqlite
import logging
from typing import List
from datetime import datetime
import hashlib

logger = logging.getLogger(__name__)

class LibrarianAgent:
    def __init__(self, db_path: str):
        self.db_path = db_path
    
    async def detect_orphans(self) -> List[str]:
        """Find notes with no backlinks"""
        async with aiosqlite.connect(self.db_path) as db:
            cursor = await db.execute("""
                SELECT vn.id FROM vault_notes vn
                WHERE NOT EXISTS (
                    SELECT 1 FROM vault_links vl WHERE vl.target_id = vn.id
                )
                ORDER BY vn.created_at DESC
            """)
            
            orphans = []
            async for (note_id,) in cursor:
                orphans.append(note_id)
            
            return orphans
    
    async def detect_stale_notes(self, days: int = 90) -> List[str]:
        """Find notes not updated recently"""
        async with aiosqlite.connect(self.db_path) as db:
            cursor = await db.execute("""
                SELECT id FROM vault_notes
                WHERE datetime(updated_at) < datetime('now', ? || ' days')
                ORDER BY updated_at DESC
            """, (f"-{days}",))
            
            stale = []
            async for (note_id,) in cursor:
                stale.append(note_id)
            
            return stale
    
    async def create_proposals(self, orphans: List[str], stale: List[str]) -> int:
        """Create proposals for orphans and stale notes"""
        async with aiosqlite.connect(self.db_path) as db:
            count = 0
            
            for note_id in orphans:
                proposal_id = hashlib.sha256(
                    f"orphan-{note_id}-{datetime.utcnow()}".encode()
                ).hexdigest()[:16]
                
                await db.execute("""
                    INSERT OR IGNORE INTO proposals 
                    (id, note_id, proposal_type, content, status)
                    VALUES (?, ?, ?, ?, ?)
                """, (proposal_id, note_id, "orphan", "Note has no backlinks", "pending"))
                count += 1
            
            for note_id in stale:
                proposal_id = hashlib.sha256(
                    f"stale-{note_id}-{datetime.utcnow()}".encode()
                ).hexdigest()[:16]
                
                await db.execute("""
                    INSERT OR IGNORE INTO proposals 
                    (id, note_id, proposal_type, content, status)
                    VALUES (?, ?, ?, ?, ?)
                """, (proposal_id, note_id, "stale", "Note hasn't been updated in 90 days", "pending"))
                count += 1
            
            await db.commit()
            return count
    
    async def run_nightly(self) -> dict:
        """Full nightly maintenance run"""
        logger.info("Librarian running nightly maintenance")
        
        try:
            orphans = await self.detect_orphans()
            stale = await self.detect_stale_notes()
            proposals = await self.create_proposals(orphans, stale)
            
            logger.info(f"Librarian: found {len(orphans)} orphans, {len(stale)} stale, created {proposals} proposals")
            
            return {
                "orphans": len(orphans),
                "stale": len(stale),
                "proposals": proposals,
            }
        except Exception as e:
            logger.error(f"Librarian error: {e}")
            return {"error": str(e)}
```

    """List all notes"""
    cursor = await db.execute("""
        SELECT id, path, title, created_at, updated_at FROM vault_notes
        ORDER BY created_at DESC
        LIMIT ?
    """, (limit,))
    
    notes = await cursor.fetchall()

@router.get("/orphans")
async def list_orphans(db = Depends(get_db)):
    """Get orphaned notes"""
    cursor = await db.execute("""
        SELECT vn.id, vn.path, vn.title FROM vault_notes vn
        WHERE NOT EXISTS (
            SELECT 1 FROM vault_links vl WHERE vl.target_id = vn.id
        )
    """)
    
    notes = await cursor.fetchall()
    return {"orphans": [dict(n) for n in notes]}

# src/nb/scheduler.py
import asyncio
import logging
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from src.nb.agents import LibrarianAgent

logger = logging.getLogger(__name__)

async def start_scheduler(db_path: str):
    """Start background task scheduler"""
    scheduler = AsyncIOScheduler()
    
    # Librarian nightly
    async def run_librarian():
        agent = LibrarianAgent(db_path)
        await agent.run_nightly()
    
    scheduler.add_job(run_librarian, "cron", hour=22, minute=0)  # 10 PM daily
    
    scheduler.start()
    logger.info("Scheduler started")
    
    return scheduler
```
